async function setup(page: Page, fixture: Fixture = { analysis: document(), result: group(), current: 1, selected: 1 }) {
  const log = { writes: [] as { path: string; body: unknown }[], unexpected: [] as string[], runtime: [] as string[], console: [] as string[] }; diagnostics.set(page, log);
  page.on('pageerror', error => log.runtime.push(error.message)); page.on('console', msg => { if (['error', 'warning'].includes(msg.type())) log.console.push(msg.text()); });
  let edited: ProjectAnalysis | null = null;
  const view = (): ProjectView => { const analysis = edited || fixture.analysis, version = edited ? 2 : fixture.selected, result = edited ? null : fixture.result, point = analysis.knowledge_points[0];
    return { project_id: 'skills-project', input, current_version: edited ? 2 : fixture.current,
      versions: Array.from({ length: edited ? 2 : fixture.current }, (_, i) => ({ version: i + 1, status: 'partial', created_at: time })),

  };
  await page.route('**/*', async route => {
    const request = route.request(), url = new URL(request.url()), origin = new URL(process.env.PLAYWRIGHT_BASE_URL || 'http://127.0.0.1:5177').origin;
    if (url.origin === origin && !url.pathname.startsWith('/api/')) return route.continue();
    if (request.method() === 'GET' && url.pathname === '/api/capabilities') return route.fulfill({ json: caps });

    if (request.method() === 'GET' && url.pathname === '/api/projects/skills-project/results') return route.fulfill({ json: view() });

}

test('phase3 三尺寸、键盘与减少动态：技能依据可展开、练习不被标成已完成', async ({ page }) => {
  await setup(page); const panel = route(page);
  await expect(panel.locator('[data-skill-id="skill_1"]')).toHaveAttribute('data-skill-state', 'content');
  await mkdir(out, { recursive: true }); const observations = [];
  for (const viewport of [{ name: 'desktop', width: 1440, height: 1000 }, { name: 'tablet', width: 768, height: 1024 }, { name: 'mobile', width: 390, height: 844 }]) {
    await page.setViewportSize({ width: viewport.width, height: viewport.height });
    expect(await page.evaluate(() => matchMedia('(prefers-reduced-motion: reduce)').matches)).toBe(true);
    const motion = await panel.evaluate(node => ({ animation: getComputedStyle(node).animationName, scroll: getComputedStyle(globalThis.document.documentElement).scrollBehavior }));
    expect(motion.animation).toBe('none'); expect(motion.scroll).not.toBe('smooth');
    expect(await page.evaluate(() => globalThis.document.documentElement.scrollWidth <= innerWidth)).toBe(true);
    const colors = await panel.evaluate(node => {
      const channels = (value: string) => value.match(/[\d.]+/g)!.map(Number);
      const lum = (rgb: number[]) => rgb.slice(0, 3).map(n => n / 255 <= .04045 ? n / 255 / 12.92 : ((n / 255 + .055) / 1.055) ** 2.4).reduce((sum, n, i) => sum + n * [.2126, .7152, .0722][i], 0);
      const bg = (e: Element): number[] => { const color = channels(getComputedStyle(e).backgroundColor); return color.length < 4 || color[3] > 0 ? color : bg(e.parentElement!); };
      return [...node.querySelectorAll('h5,h6,p,strong,.skill-support-state,.practice-state')].filter(e => e.checkVisibility()).map(e => { const a = lum(channels(getComputedStyle(e).color)), b = lum(bg(e)); return { text: e.textContent?.trim(), ratio: (Math.max(a, b) + .05) / (Math.min(a, b) + .05) }; });
    });
    expect(colors.every(row => row.ratio >= 4.5), JSON.stringify(colors)).toBe(true);
    await page.screenshot({ path: resolve(out, `two-axes-${viewport.name}.png`), fullPage: true, animations: 'disabled' });
    observations.push({ viewport, overflow: false, reducedMotion: motion, colors });
  }
  const summary = panel.locator('[data-skill-id="skill_1"] summary'); await summary.focus(); await page.keyboard.press('Enter');
  await expect(panel.locator('[data-skill-id="skill_1"] details')).toHaveAttribute('open', '');
  await expect(panel.locator('[data-skill-id="skill_1"] blockquote')).toContainText('解析日期时间');
  await page.keyboard.press('Tab'); await expect(panel.locator('[data-skill-id="skill_1"] a')).toBeFocused();
  await expect(panel.locator('.practice-state')).toHaveText('待自己实践');
  await writeFile(resolve(out, 'responsive-observations.json'), JSON.stringify({ mock: true, viewports: observations, keyboardEvidenceDetailsOpened: true, sourcePlaybackOrTeachingQualityAccepted: false }, null, 2));
});
