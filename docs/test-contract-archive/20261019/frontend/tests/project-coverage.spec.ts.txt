async function setup(page: Page, groups: Map<number, ResultGroup | null>, status = 'completed', document = analysis, projectInput = input, pointState?: { current: string }, mockImageUrls: string[] = []) {
  const log: Logs = { runtime: [], console: [], unexpected: [] }; diagnostics.set(page, log);
  page.on('pageerror', error => log.runtime.push(error.message));
  page.on('console', message => { if (['warning', 'error'].includes(message.type())) log.console.push(message.text()); });
  const current = Math.max(...groups.keys());
  await page.route('**/*', route => {
    const request = route.request(); const url = new URL(request.url());
    const origin = new URL(process.env.PLAYWRIGHT_BASE_URL || 'http://127.0.0.1:5173').origin;
    if (url.origin === origin && !url.pathname.startsWith('/api/')) return route.continue();
    if (request.method() === 'GET' && url.pathname === '/api/capabilities') return route.fulfill({ json: caps });
    if (request.method() === 'GET' && url.pathname === '/api/projects') return route.fulfill({ json: { projects: [{ project_id: 'coverage-project', name: projectInput.name, current_version: current, created_at: time, updated_at: time }] } });
    if (request.method() === 'GET' && url.pathname === '/api/projects/coverage-project/results') {
      const selected = Number(url.searchParams.get('version')) || current; const result = groups.get(selected);
      const view: ProjectView = { project_id: 'coverage-project', input: projectInput, current_version: current,
        versions: [...groups.keys()].map(version => ({ version, status, created_at: time })),
        version: { version: selected, status, input: projectInput, analysis: document, confirmation: { version: selected, confirmed_at: time },
          batches: [{ batch_id: `batch-${selected}`, status, project_knowledge_point_ids: [document.knowledge_points[0].id], job_id: `job-${selected}` }], error: null, created_at: time },
        points: [{ project_knowledge_point_id: document.knowledge_points[0].id, name: document.knowledge_points[0].name, selection: 'learn', status: pointState?.current ?? (status === 'running' ? 'running' : result?.items.length ? 'completed' : result?.partial_items?.length || result?.review_items?.length ? 'partial' : 'no_results'),
          batch_id: `batch-${selected}`, job_id: `job-${selected}`, knowledge_point_id: result?.knowledge_point_id || null, result: result || null }] };
      return route.fulfill({ json: view });
    }
    if (request.resourceType() === 'image' && mockImageUrls.includes(url.href)) return route.fulfill({ contentType: 'image/png', body: Buffer.from('iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mP8/x8AAwMCAO+jrZkAAAAASUVORK5CYII=', 'base64') });
    log.unexpected.push(`${request.method()} ${url.href}`); return route.abort('blockedbyclient');
  });
  await page.goto(`/?project=coverage-project&version=${current}`);
  await expect(page).toHaveTitle('项目制知识引导小精灵 · 课程与视频发现');
  await expect(page.getByRole('heading', { name: `项目课程结果 · v${current}` })).toBeVisible();
  await expect(page.locator('vite-error-overlay')).toHaveCount(0);
  await openLegacy(page);
  return log;
}

async function contrast(page: Page) {
  return page.locator('.unit-coverage').evaluate(container => {
    const channels = (value: string) => value.match(/[\d.]+/g)!.map(Number);
    const luminance = (rgb: number[]) => rgb.slice(0, 3).map(n => n / 255 <= .04045 ? n / 255 / 12.92 : ((n / 255 + .055) / 1.055) ** 2.4).reduce((sum, n, i) => sum + n * [.2126,.7152,.0722][i], 0);
    const background = (element: Element): number[] => { const c = channels(getComputedStyle(element).backgroundColor); return c.length < 4 || c[3] > 0 ? c : background(element.parentElement!); };
    return [...container.querySelectorAll('h5,h6,li>span,li>small,.unit-coverage-supported p,.unit-coverage-missing li,.unit-coverage-next p,.unit-coverage-next a,.unit-coverage-practice span,.unit-coverage-note,.unit-coverage-heading>span')].filter(e => e.checkVisibility()).map(element => {
      const style = getComputedStyle(element); const a = luminance(channels(style.color)); const b = luminance(background(element));
      return { text: element.textContent?.trim(), ratio: (Math.max(a,b)+.05)/(Math.min(a,b)+.05), fontSize: style.fontSize };
    });
  });
}
