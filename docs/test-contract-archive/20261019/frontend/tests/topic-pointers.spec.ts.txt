async function setup(page: Page, f = fixture()) {

  page.on('pageerror', error => log.runtime.push(error.message)); page.on('console', msg => { if (['error','warning'].includes(msg.type())) log.console.push(msg.text()); });
  const view: ProjectView = { project_id: 'pointer-project', input, current_version: 1, versions: [{ version: 1, status: 'partial', created_at: time }],
    version: { version: 1, status: 'partial', input, analysis: f.analysis, confirmation: { confirmed_at: time }, batches: [{ batch_id: 'batch1', job_id: 'job1', status: 'completed', project_knowledge_point_ids: ['k1'] }], error: null, created_at: time },

  const origin = new URL(process.env.PLAYWRIGHT_BASE_URL || 'http://127.0.0.1:5178').origin;
  await page.route('**/*', async route => {
    const request=route.request(), url=new URL(request.url());
    if(request.method()!=='GET'){log.writes.push(`${request.method()} ${url.href}`);return route.abort('blockedbyclient');}
    if(url.origin===origin && !url.pathname.startsWith('/api/')) return route.continue();

test('three widths and reduce motion: folded/expanded pointer long source stays readable and keyboard reachable',async({page})=>{
  await setup(page);await mkdir(out,{recursive:true});const observations=[];
  for(const size of [{name:'desktop',width:1440,height:1000},{name:'tablet',width:768,height:1024},{name:'mobile',width:390,height:844}]){
    await page.setViewportSize(size);await fold(page).evaluate(node=>(node as HTMLDetailsElement).open=false);await fold(page).scrollIntoViewIfNeeded();
    await page.screenshot({path:resolve(out,`${size.name}-folded.png`),animations:'disabled'});
    await fold(page).locator('summary').focus();await page.keyboard.press('Enter');await expect(fold(page)).toHaveAttribute('open','');

    const overflow=await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth+1);expect(overflow).toBe(false);
    // Screen-reader text is intentionally clipped to1px; visible text remains
    // subject to the full interior-overflow check.
    const inner=await fold(page).evaluate(node=>[...node.querySelectorAll('*')].filter(el=>!el.classList.contains('sr-only')&&el.clientWidth>0&&el.scrollWidth>el.clientWidth+1).map(el=>el.tagName));expect(inner).toEqual([]);
    const motions=await fold(page).evaluate(node=>({animation:getComputedStyle(node).animationName,transition:getComputedStyle(node).transitionDuration}));expect(motions).toEqual({animation:'none',transition:'0s'});
    await page.keyboard.press('Tab');await expect(fold(page).getByRole('link')).toBeFocused();observations.push({size,overflow,inner,motions});
  }
  await writeFile(resolve(out,'three-width-observations.json'),JSON.stringify({mock:true,actualRetainedCatalog:true,noSourceSemanticAcceptance:true,observations},null,2));
